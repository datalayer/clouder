```bash
cat << 'EOF' > /tmp/values.yaml
...
EOF
export RELEASE=datalayer-outbounds
export NAMESPACE=datalayer-outbounds
helm upgrade \
    --install $RELEASE \
    datalayer/outbounds \
    --create-namespace \
    --namespace $NAMESPACE \
    --values /tmp/values.yaml \
    --timeout 10m
```
